pkg_check_modules(ADW IMPORTED_TARGET libadwaita-1 gtk4 fontconfig)
if(ADW_FOUND)
  add_executable(kuockey-settings Settings.cpp Guide.cpp Emoji.cpp ../ibus/Prefs.cpp)
  target_compile_definitions(kuockey-settings PRIVATE KUOCKEY_VERSION="${KUOCKEY_VERSION}" KUOCKEY_DATADIR="${CMAKE_INSTALL_FULL_DATADIR}/kuockey")
  target_link_libraries(kuockey-settings PRIVATE PkgConfig::ADW)
  pkg_check_modules(GTKX11 IMPORTED_TARGET gtk4-x11 x11)
  if(GTKX11_FOUND)   # the emoji picker places itself in the middle of the screen (X11; Wayland has no window positions)
    target_compile_definitions(kuockey-settings PRIVATE KUOCKEY_HAVE_X11)
    target_link_libraries(kuockey-settings PRIVATE PkgConfig::GTKX11)
  endif()
  install(TARGETS kuockey-settings DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES kuockey.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES ../icons/icon.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME kuockey.png)
else()
  message(WARNING "libadwaita not found: kuockey-settings not built")
endif()
