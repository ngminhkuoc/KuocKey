cmake_minimum_required(VERSION 3.22)
project(kuockey LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ENGINE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../Engine)
add_library(kuockey SHARED kuockey_jni.cpp ${ENGINE}/Engine.cpp ${ENGINE}/Vietnamese.cpp ${ENGINE}/SmartSwitchKey.cpp ${ENGINE}/Driver.cpp ${ENGINE}/TouchModel.cpp)
target_include_directories(kuockey PRIVATE ${ENGINE} ${ENGINE}/platforms)
target_compile_options(kuockey PRIVATE -w -O2 -fvisibility=hidden)
target_link_libraries(kuockey log)
# 16 KB memory pages (Android 15+ devices; required by Google Play): align the ELF segments to 16 KB
target_link_options(kuockey PRIVATE "-Wl,-z,max-page-size=16384")
